\chapter*{Formula Sheet}
\addcontentsline{toc}{chapter}{Formula Sheet}
\markboth{Formula Sheet}{Formula Sheet}

The ASVAB does not give you formulas. Everything on this page is fair
game---learn it until you can write it from memory.

{\small
\tcbset{before skip=3pt,after skip=5pt,top=2pt,bottom=3pt,left=5pt,right=5pt}
\setlist{itemsep=0pt,topsep=1pt,parsep=0pt}
\begin{multicols}{2}
\begin{concept}{Percents \& money}
\begin{itemize}[leftmargin=1em]
\item part $=$ percent $\times$ whole
\item percent change $= \dfrac{\text{new}-\text{original}}{\text{original}}\times 100\%$
\item sale price $=$ price $\times (1 - \text{discount rate})$
\item total with tax $=$ price $\times (1 + \text{tax rate})$
\item simple interest: $I = Prt$ \ ($t$ in years)
\end{itemize}
\end{concept}

\begin{concept}{Rates}
\begin{itemize}[leftmargin=1em]
\item distance $=$ rate $\times$ time, \ $d = rt$
\item average speed $= \dfrac{\text{total distance}}{\text{total time}}$
\item work together: $\dfrac1a + \dfrac1b = \dfrac1t$
\item unit rate $= \dfrac{\text{amount}}{\text{one unit}}$
\end{itemize}
\end{concept}

\begin{concept}{Statistics \& probability}
\begin{itemize}[leftmargin=1em]
\item mean $= \dfrac{\text{sum of values}}{\text{number of values}}$
\item median: middle value of the \emph{sorted} list
\item mode: most frequent value; \ range $=$ max $-$ min
\item $P(\text{event}) = \dfrac{\text{favorable outcomes}}{\text{total outcomes}}$
\item $P(\text{not } A) = 1 - P(A)$; \ independent: $P(A \text{ and } B) = P(A)\,P(B)$
\end{itemize}
\end{concept}

\begin{concept}{Exponents \& roots}
\begin{itemize}[leftmargin=1em]
\item $a^m \cdot a^n = a^{m+n}$, \ $\dfrac{a^m}{a^n} = a^{m-n}$, \ $(a^m)^n = a^{mn}$
\item $a^0 = 1$, \ $a^{-n} = \dfrac{1}{a^n}$, \ $(ab)^n = a^n b^n$
\item $\sqrt{ab} = \sqrt a\,\sqrt b$, \ $\sqrt{\dfrac ab} = \dfrac{\sqrt a}{\sqrt b}$
\end{itemize}
\end{concept}

\begin{concept}{Algebra}
\begin{itemize}[leftmargin=1em]
\item $(a+b)^2 = a^2 + 2ab + b^2$
\item $(a-b)^2 = a^2 - 2ab + b^2$
\item $(a+b)(a-b) = a^2 - b^2$
\item quadratic formula: $x = \dfrac{-b \pm \sqrt{b^2 - 4ac}}{2a}$
\item slope $m = \dfrac{y_2 - y_1}{x_2 - x_1}$; \ line: $y = mx + b$
\item distance $= \sqrt{(x_2-x_1)^2 + (y_2-y_1)^2}$
\item midpoint $= \left(\dfrac{x_1+x_2}{2}, \dfrac{y_1+y_2}{2}\right)$
\end{itemize}
\end{concept}

\begin{concept}{Angles \& triangles}
\begin{itemize}[leftmargin=1em]
\item complementary: sum $90^\circ$; \ supplementary: sum $180^\circ$
\item angles of a triangle add to $180^\circ$
\item polygon with $n$ sides: angle sum $(n-2)\times 180^\circ$
\item Pythagorean theorem: $a^2 + b^2 = c^2$
\item common triples: 3-4-5, 5-12-13, 8-15-17, 7-24-25
\item 45-45-90: sides $x,\ x,\ x\sqrt2$; \ 30-60-90: $x,\ x\sqrt3,\ 2x$
\end{itemize}
\end{concept}

\begin{concept}{Perimeter \& area}
\begin{itemize}[leftmargin=1em]
\item rectangle: $P = 2l + 2w$, \ $A = lw$
\item square: $P = 4s$, \ $A = s^2$
\item triangle: $A = \frac12 bh$
\item parallelogram: $A = bh$ \ (height, not slanted side)
\item trapezoid: $A = \frac12(b_1 + b_2)h$
\end{itemize}
\end{concept}

\begin{concept}{Circles \& solids}
\begin{itemize}[leftmargin=1em]
\item $d = 2r$, \ circumference $C = 2\pi r = \pi d$
\item area $A = \pi r^2$; \ $\pi \approx 3.14 \approx \frac{22}{7}$
\item box: $V = lwh$, \ $SA = 2lw + 2lh + 2wh$
\item cube: $V = s^3$, \ $SA = 6s^2$
\item cylinder: $V = \pi r^2 h$
\item cone: $V = \frac13\pi r^2 h$; \ sphere: $V = \frac43\pi r^3$
\end{itemize}
\end{concept}
\end{multicols}
}

\subsection*{Units of measure}
\begin{tabular}{@{}l@{}}
12 inches = 1 foot; \ 3 feet = 1 yard\\
5,280 feet = 1 mile; \ 1,760 yards = 1 mile\\
8 fluid ounces = 1 cup; \ 2 cups = 1 pint\\
2 pints = 1 quart; \ 4 quarts = 1 gallon\\
16 ounces = 1 pound; \ 2,000 pounds = 1 ton\\
60 seconds = 1 minute; \ 60 minutes = 1 hour\\
1 yard$^2$ = 9 feet$^2$; \ 1 foot$^2$ = 144 inches$^2$\\
1 yard$^3$ = 27 feet$^3$\\[3pt]
\textbf{Metric:} kilo = 1,000; centi = $\frac{1}{100}$; milli = $\frac{1}{1000}$\\
1 km = 1,000 m; \ 1 m = 100 cm = 1,000 mm\\
1 kg = 1,000 g; \ 1 L = 1,000 mL
\end{tabular}

\subsection*{24-hour (military) time}
Hours after noon: add 12. \ 1:00~p.m. = 1300, \ 6:45~p.m. = 1845,
\ midnight = 0000, \ 12:30~a.m. = 0030.
