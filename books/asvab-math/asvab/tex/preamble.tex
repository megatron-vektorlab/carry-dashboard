\newcommand{\choicelabel}[1]{{\sans\bfseries #1}\enspace}
\newlength{\chw}\newlength{\chmax}
\newcommand{\measurechoice}[1]{\settowidth{\chw}{\choicelabel{A}#1}%
  \ifdim\chw>\chmax\global\setlength{\chmax}{\chw}\fi}
% \choices{A}{B}{C}{D}: four across, two-by-two, or one per line
\newcommand{\choices}[4]{%
  \global\setlength{\chmax}{0pt}%
  \measurechoice{#1}\measurechoice{#2}\measurechoice{#3}\measurechoice{#4}%
  \par\nopagebreak\vspace{1pt}\noindent
  \ifdim\chmax<0.235\linewidth
    \makebox[.25\linewidth][l]{\choicelabel{A}#1}%
    \makebox[.25\linewidth][l]{\choicelabel{B}#2}%
    \makebox[.25\linewidth][l]{\choicelabel{C}#3}%
    \makebox[.25\linewidth][l]{\choicelabel{D}#4}%
  \else\ifdim\chmax<0.48\linewidth
    \makebox[.5\linewidth][l]{\choicelabel{A}#1}%
    \makebox[.5\linewidth][l]{\choicelabel{B}#2}\par\noindent
    \makebox[.5\linewidth][l]{\choicelabel{C}#3}%
    \makebox[.5\linewidth][l]{\choicelabel{D}#4}%
  \else
    \begin{tabular}{@{}l@{}p{\dimexpr\linewidth-1.4em}@{}}
      \choicelabel{A} & #1\\ \choicelabel{B} & #2\\
      \choicelabel{C} & #3\\ \choicelabel{D} & #4
    \end{tabular}%
  \fi\fi\par}
